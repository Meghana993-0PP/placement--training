class Solution {
    public int addDigits(int num) {      
        while(num>=10) // outer while
	{
         int sum=0;
         while(num>0) // inner while
         {
         sum=sum+(num%10); // 8
         num=num/10; // 3
	}
      num=sum;
    }
    return num;
}
}

342. Power of Four

class Solution {
    public boolean isPowerOfFour(int n) {
       if(n<=0)
         return false;
        while(n>1)
        {
            if(n%4!=0)
                return false;
            n=n/4;
        }  
        return true;
    }
}

class Solution {
    public int subtractProductAndSum(int n) {
        int sum=0, pro=1;
      while(n>0)
      {
      int digit = n%10;
       sum+=digit;
       pro*=digit;
       n=n/10;
       }
 return pro-sum;
    }

1323. Maximum 69 Number
class Solution {
    public int maximum69Number (int num) {
        char[] c=String.valueOf(num).toCharArray();
        for(int i=0;i<c.length;i++)
        {
            if(c[i]=='6')
            {
                c[i]='9';
                break;
            }
        }
        return Integer.parseInt(new String(c));
    }
}
